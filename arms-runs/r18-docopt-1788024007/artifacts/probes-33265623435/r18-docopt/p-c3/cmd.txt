import json, os, re, urllib.request, urllib.error
def get(url, hdr={}):
    try:
        r = urllib.request.urlopen(urllib.request.Request(url, headers={'User-Agent': 'probe', **hdr}), timeout=30)
        return r.status, r.read().decode(errors='replace')
    except urllib.error.HTTPError as e:
        return e.code, e.read().decode(errors='replace')[:200]
    except Exception as e:
        return None, repr(e)
want = {'2.7', '3.4', '3.5', '3.6'}
travis = open('.travis.yml').read() if os.path.exists('.travis.yml') else ''
pys = set(re.findall(r'^\s*-\s*["\']?(\d\.\d+)["\']?\s*$', travis, re.M))
local_wf = os.listdir('.github/workflows') if os.path.isdir('.github/workflows') else []
print('observed: .travis.yml present =', bool(travis), '| python versions in .travis.yml =', sorted(pys), '| local GH workflows =', local_wf, '| .travis.yml head =', repr(travis[:200]))
s1, b1 = get('https://api.github.com/repos/docopt/docopt/actions/runs?per_page=1')
gha_runs = json.loads(b1).get('total_count') if s1 == 200 else (0 if not local_wf else None)
