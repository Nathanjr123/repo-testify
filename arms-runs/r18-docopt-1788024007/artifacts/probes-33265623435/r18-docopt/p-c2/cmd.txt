import ast, sys, subprocess
src = open('docopt.py').read()
mods = set()
for n in ast.walk(ast.parse(src)):
    if isinstance(n, ast.Import):
        mods.update(a.name.split('.')[0] for a in n.names)
    elif isinstance(n, ast.ImportFrom) and n.module:
        mods.add(n.module.split('.')[0])
non_std = sorted(m for m in mods if m not in sys.stdlib_module_names)
print('observed: imports =', sorted(mods), '| non-stdlib =', non_std, '| lines =', src.count(chr(10)))
p = subprocess.run([sys.executable, '-c', "from docopt import docopt; print(dict(docopt('Usage: x [-v] <f>', argv=['-v','a'])))"], capture_output=True, text=True, cwd='/tmp/dropin')
print('observed: standalone run exit', p.returncode, '| stdout', p.stdout.strip(), '| stderr', p.stderr.strip()[-200:])
assert not non_std, non_std
assert p.returncode == 0 and "'-v': True" in p.stdout and "'<f>': 'a'" in p.stdout, p.stdout
