url = 'https://travis-ci.org/docopt/docopt.svg?branch=master'

    r = urllib.request.urlopen(urllib.request.Request(url, headers={'User-Agent': 'probe'}), timeout=30)

    print('VERDICT_LINE: PASS Travis badge for master renders build passing')

    print('VERDICT_LINE: FAIL Travis badge for master renders %s, not passing' % (texts[:8],))

    print('VERDICT_LINE: FAIL Travis badge unreachable: http %s at %s' % (status, final))
